/* 1 */
SELECT CUS_LNAME, CUS_FNAME, INV_NUMBER, INV_DATE
FROM CUSTOMER INNER JOIN INVOICE ON CUSTOMER.CUS_CODE = INVOICE.CUS_CODE;

/* 2 */
SELECT INV_NUMBER, LINE_NUMBER, LINE.P_CODE
FROM LINE INNER JOIN PRODUCT ON LINE.P_CODE = PRODUCT.P_CODE;

/* 3 */
SELECT INV_NUMBER, LINE.P_CODE, PRODUCT.P_DESCRIPT, LINE_PRICE
FROM LINE INNER JOIN PRODUCT ON LINE.P_CODE = PRODUCT.P_CODE;

/* 4 */
SELECT P_DESCRIPT, P_PRICE, V_NAME
FROM PRODUCT INNER JOIN VENDOR ON PRODUCT.V_CODE = VENDOR.V_CODE;

/* 5 */
SELECT V_NAME, P_CODE
FROM VENDOR INNER JOIN PRODUCT ON VENDOR.V_CODE = PRODUCT.V_CODE;
